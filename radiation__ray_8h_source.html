151ac_ray_func(int3 direction, Field kappa_rho, Field srad, Field Q, Field tau, real in_len, real out_len)
152{
153 dtau_m = sqrt(incoming_ray(kappa_rho,direction)*kappa_rho)*in_len
154 dtau_p = sqrt(outgoing_ray(kappa_rho,direction)*kappa_rho)*out_len
155
156 dtau_m = max(dtau_m,AC_REAL_EPSILON*5)
157 dtau_p = max(dtau_p,AC_REAL_EPSILON*5)
158
159 dSdtau_m = (srad - incoming_ray(srad,direction))/dtau_m
160 dSdtau_p = (outgoing_ray(srad,direction)- srad)/dtau_p
161
162 Srad1st=(dSdtau_p*dtau_m+dSdtau_m*dtau_p)/(dtau_m+dtau_p)
163 Srad2nd=2.0*(dSdtau_p-dSdtau_m)/(dtau_m+dtau_p)
164
165 write(tau,incoming_ray(tau,direction)+dtau_m)
166 real emdtau
167 real emdtau1
168 real emdtau2
169 if(dtau_m > dtau_thresh_max)
170 {
171 emdtau=0.0
172 emdtau1=1.0
173 emdtau2=-1.0
174 }
175 else if(dtau_m < dtau_thresh_min)
176 {
177 emdtau1=dtau_m*(1.0-0.5*dtau_m*(1.0-dtau_m/3.0))
178 emdtau=1-emdtau1
179 emdtau2=-(dtau_m*dtau_m)*(0.5-dtau_m/3.0)
180 }
181 else
182 {
183 emdtau=exp(-dtau_m)
184 emdtau1=1.0-emdtau
185 emdtau2=emdtau*(1.0+dtau_m)-1.0
186 }
187 const Q_res = incoming_ray(Q,direction)*emdtau-Srad1st*emdtau1-Srad2nd*emdtau2
188 write(Q,Q_res)
189}
