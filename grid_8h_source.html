117AcReal
118ac_get_normalized_plm(const AcReal x, const int l, const int m)
119{
120 const AcReal plm = ac_get_plm(x,l,m);
121 const AcReal fact_1 = AcReal(factorial(l-m));
122 const AcReal fact_2 = AcReal(factorial(l+m));
123 AcReal res = sqrt(((2.0*l+1.0)/(4.0*AC_REAL_PI))*(fact_1/fact_2))*plm;
124 if(m != 0) res *= sqrt(2.0);
125 //Condon-Shortley phase term which is not included in the stdlib implementation
126 if(m != 0 && m %2 == 1) res *= -1;
127 return res;
128}

397AcReal4
398ac_get_power_mapping(const AcReal xi_scaled,const AcReal exponent)
399{
400 return
401 {
402 pow(xi_scaled,1.0/exponent),
403 1.0/exponent * pow(xi_scaled,(1.0/exponent)-1.0),
404 1.0/exponent * ((1.0/exponent)-1.0)*pow(xi_scaled,(1/exponent)-2.0),
405 1.0/exponent * ((1.0/exponent)-1.0) * ((1.0/exponent)-2.0) * pow(xi_scaled,(1/exponent)-3.0)
406 };
407}
