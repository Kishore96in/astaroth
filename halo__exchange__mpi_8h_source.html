164template <typename T, typename Allocator> class halo_exchange {
165 private:
166 std::vector<std::unique_ptr<packet<T, Allocator>>> m_packets;
167
168 public:
169 halo_exchange(const MPI_Comm& parent_comm, const ac::shape& global_nn, const ac::index& rr)
170 {
171 // Get local mesh dimensions
172 const auto local_mm{ac::mpi::get_local_mm(parent_comm, global_nn, rr)};
173 const auto local_nn{ac::mpi::get_local_nn(parent_comm, global_nn)};
174
175 // Partition the domain
176 auto segments{partition(local_mm, local_nn, rr)};
177
178 // Prune the segment containing the computational domain
179 for (size_t i{0}; i < segments.size(); ++i) {
180 if (within_box(segments[i].offset, local_nn, rr)) {
181 segments.erase(segments.begin() + as<long>(i));
182 --i;
183 }
184 }
185
186 // Sort the packets from largest to smallest
187 std::sort(segments.begin(), segments.end(), [](const auto& a, const auto& b) {
188 return ac::prod(a.dims) > ac::prod(b.dims);
189 });
190
191 for (const auto& segment : segments)
192 m_packets.push_back(
193 std::make_unique<packet<T, Allocator>>(parent_comm, global_nn, rr, segment));
194 }
195
196 void launch(const ac::view<T, Allocator>& input, ac::view<T, Allocator> output)
197 {
198 ERRCHK_MPI(complete());
199
200 for (auto& packet : m_packets)
201 packet->launch(input, output);
202 }
203
205 void wait_one()
206 {
207 ERRCHK_MPI(!complete());
208
209 for (auto& packet : m_packets)
210 if (!packet->complete() && packet->ready())
211 packet->wait();
212 }
213
215 void wait_all()
216 {
217 ERRCHK_MPI(!complete());
218
219 while (!complete())
220 wait_one();
221 }
222
224 bool ready() const
225 {
226 ERRCHK_MPI(!complete());
227
228 for (const auto& packet : m_packets)
229 if (!packet->complete() && packet->ready())
230 return true;
231
232 return false;
233 }
234
236 bool complete() const
237 {
238 for (const auto& packet : m_packets)
239 if (!packet->complete())

241
