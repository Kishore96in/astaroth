62 template <typename Allocator>
63 void launch_write_collective(const MPI_Comm& parent_comm, const ac::view<T, Allocator>& input,
64 const std::string& path)
65 {
66 ERRCHK_MPI(!m_in_progress);
67 m_in_progress = true;
68
69 // Communicator
70 ERRCHK_MPI(m_comm == MPI_COMM_NULL);
71 ERRCHK_MPI_API(MPI_Comm_dup(parent_comm, &m_comm));
72
73 // TODO: transfers the whole buffer at the moment, would be
74 // better to migrate only mesh_subdims instead (but need
75 // to pack and change mesh_offset to zero)
76 // migrate(input, m_staging_buffer);
77 ac::copy<T>(input, m_staging_buffer.get());
78
79 ERRCHK_MPI(m_file == MPI_FILE_NULL);
80 ERRCHK_MPI_API(MPI_File_open(m_comm,
81 path.c_str(),
82 MPI_MODE_CREATE | MPI_MODE_WRONLY,
83 m_info,
84 &m_file));
85 ERRCHK_MPI_API(MPI_File_set_view(m_file,
86 0,
87 ac::mpi::get_dtype<T>(),
88 m_global_subarray,
89 "native",
90 m_info));
91
92 ERRCHK_MPI(m_file != MPI_FILE_NULL);
93 ERRCHK_MPI(m_req == MPI_REQUEST_NULL);
94 ERRCHK_MPI_API(
95 MPI_File_iwrite_all(m_file, m_staging_buffer.data(), 1, m_local_subarray, &m_req));
96 }

98 void wait_write_collective()
99 {
100 ERRCHK_MPI(m_in_progress);
101 ERRCHK_MPI_API(MPI_Wait(&m_req, MPI_STATUS_IGNORE));
102 ERRCHK_MPI_API(MPI_File_close(&m_file));
103
104 // Ensure all processess have written their segment to disk.
105 // Not strictly needed here and comes with a slight overhead
106 // However, will cause hard-to-trace issues if the reader
107 // tries to access the data without barrier.
108 ERRCHK_MPI(m_comm != MPI_COMM_NULL);
109 ERRCHK_MPI_API(MPI_Barrier(m_comm));
110 ERRCHK_MPI_API(MPI_Comm_free(&m_comm));
111
112 // Check that the MPI implementation reset the resources
113 ERRCHK_MPI(m_req == MPI_REQUEST_NULL);
114 ERRCHK_MPI(m_file == MPI_FILE_NULL);
115 ERRCHK_MPI(m_comm == MPI_COMM_NULL);
116
117 // Complete
118 m_in_progress = false;
119 }

152 const std::vector<std::string>& paths)
