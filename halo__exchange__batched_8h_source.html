51 void launch(const ac::view<T, Allocator>& input, ac::view<T, Allocator> output)
52 {
53 ERRCHK_MPI(m_send_req == MPI_REQUEST_NULL);
54 ERRCHK_MPI(m_recv_req == MPI_REQUEST_NULL);
55 ERRCHK_MPI(input.size() == output.size());
56
57 if (const auto rank{ac::mpi::get_rank(m_comm)};
58 rank == m_recv_neighbor && rank == m_send_neighbor) {
59 // Pack directly to the output buffer if the destination is on the same device
60 ac::copy(input, output); // Can be improved by using async memcpy
61
62 // Create a dummy request to simplify asynchronous management
63 static int dummy{-1};
64 ERRCHK_MPI_API(MPI_Irecv(&dummy, 0, MPI_INT, rank, m_tag, m_comm, &m_recv_req));
65 ERRCHK_MPI_API(MPI_Isend(&dummy, 0, MPI_INT, rank, m_tag, m_comm, &m_send_req));
66 }
67 else { // Use MPI communication
68 ERRCHK_MPI_API(MPI_Irecv(output.data(),
69 as<int>(output.size()),
70 ac::mpi::get_dtype<T>(),
71 m_recv_neighbor,
72 m_tag,
73 m_comm,
74 &m_recv_req));
75
76 ERRCHK_MPI_API(MPI_Isend(input.data(),
77 as<int>(input.size()),
78 ac::mpi::get_dtype<T>(),
79 m_send_neighbor,
80 m_tag,
81 m_comm,
82 &m_send_req));
83 }
84
85 ac::mpi::increment_tag(m_tag);
86 }

119template <typename T, typename Allocator> class halo_exchange {
120 private:
121 ac::shape m_local_mm;
122
123 std::vector<ac::segment> m_recv_segments;
124 std::vector<ac::segment> m_send_segments;
125
126 std::vector<std::unique_ptr<ac::buffer<T, Allocator>>> m_recv_buffers;
127 std::vector<std::unique_ptr<ac::buffer<T, Allocator>>> m_send_buffers;
128
129 std::vector<std::unique_ptr<packet<T, Allocator>>> m_packets;
130
131 public:
132 halo_exchange() = default;
133
134 halo_exchange(const MPI_Comm& parent_comm, const ac::shape& global_nn, const ac::index& rr,
135 const size_t n_max_aggregate_buffers)
136 : m_local_mm{ac::mpi::get_local_mm(parent_comm, global_nn, rr)}
137 {
138 // Get local mesh dimensions
139 const auto local_nn{ac::mpi::get_local_nn(parent_comm, global_nn)};
140
141 // Partition the domain
142 auto segments{partition(m_local_mm, local_nn, rr)};
143
144 // Prune the segment containing the computational domain
145 segments = prune(segments, local_nn, rr);
146
147 // Sort the packets from largest to smallest
148 std::sort(segments.begin(), segments.end(), [](const auto& a, const auto& b) {
149 return ac::prod(a.dims) > ac::prod(b.dims);
150 });
151
152 for (const auto& segment : segments) {
153
154 const ac::segment recv_segment{segment};
155 const ac::segment send_segment{recv_segment.dims,
156 ((local_nn + recv_segment.offset - rr) % local_nn) + rr};
157
158 const ac::direction recv_direction{
159 ac::mpi::get_direction(recv_segment.offset, local_nn, rr)};
160 const auto recv_neighbor{ac::mpi::get_neighbor(parent_comm, recv_direction)};
161 const auto send_neighbor{ac::mpi::get_neighbor(parent_comm, -recv_direction)};
162
163 m_recv_segments.push_back(recv_segment);
164 m_send_segments.push_back(send_segment);
165
166 m_recv_buffers.push_back(std::make_unique<ac::buffer<T, Allocator>>(
167 n_max_aggregate_buffers * prod(recv_segment.dims)));
168 m_send_buffers.push_back(std::make_unique<ac::buffer<T, Allocator>>(
169 n_max_aggregate_buffers * prod(send_segment.dims)));
170
171 m_packets.push_back(
172 std::make_unique<packet<T, Allocator>>(parent_comm, recv_neighbor, send_neighbor));
173 }
174 }
175
176 void launch(const std::vector<ac::view<T, Allocator>>& inputs)
177 {
178 ERRCHK_MPI(complete());
179
180 pack_batched(m_local_mm, inputs, m_send_segments, unwrap_ptr_get(m_send_buffers));
181
182 for (size_t i{0}; i < m_packets.size(); ++i)
183 m_packets[i]->launch(m_send_buffers[i]->get(), m_recv_buffers[i]->get());
184 }
185
186 void wait(std::vector<ac::view<T, Allocator>> outputs)
187 {
188 ERRCHK_MPI(!complete());
189
190 for (auto& packet : m_packets)
191 packet->wait();
192
193 unpack_batched(m_recv_segments, unwrap_ptr_get(m_recv_buffers), m_local_mm, outputs);
194 }
195
197 bool complete() const
198 {
199 for (const auto& packet : m_packets)
200 if (!packet->complete())
201 return false;
202
203 return true;
204 }
205};
