156template <typename T>
157constexpr MPI_Datatype
158get_dtype()
159{
160 if constexpr (std::is_same_v<T, double>) {
161 return MPI_DOUBLE;
162 }
163 else if constexpr (std::is_same_v<T, float>) {
164 return MPI_DOUBLE;
165 }
166 else if constexpr (std::is_same_v<T, int>) {
167 return MPI_INT;
168 }
169 else if constexpr (std::is_same_v<T, uint64_t>) {
170 return MPI_UINT64_T;
171 }
172 else {
173 ERROR_DESC("Unknown datatype");
174 return MPI_DATATYPE_NULL;
175 }
176}
