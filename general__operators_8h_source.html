9u_dot_grad(Field3 f, Matrix m,real3 u){
10 suppress_unused_warning(f)
11 real3 res = m*u
12 if(AC_coordinate_system == AC_SPHERICAL_COORDINATES)
13 {
14 res.x = res.x - AC_INV_R*(u.y*f.y+u.z*f.z)
15 res.y = res.y + AC_INV_R*(u.y*f.x-u.z*f.z*AC_COT)
16 res.z = res.z + AC_INV_R*(u.z*f.x+u.z*f.y*AC_COT)
17 }
18 else if(AC_coordinate_system == AC_CYLINDRICAL_COORDINATES)
19 {
20 res.x = res.x - AC_INV_CYL_R*(u.y*f.y)
21 res.y = res.y + AC_INV_CYL_R*(u.y*f.x)
22 }
23 return res;
24}

760 )

772 )
773 +AC_COT*dery(s.x)
774 )

776 2*(

797traceless_strain(Matrix uij,divu)
798{
799 if(AC_coordinate_system == AC_SPHERICAL_COORDINATES || AC_coordinate_system == AC_CYLINDRICAL_COORDINATES)
800 {
801 fatal_error_message(true,"traceless_strain needs the velocity for curvilinear coordinates");
802 }
803 Matrix sij
804 for row in 0:3{
805 sij[row][row] = uij[row][row] - (1.0/3.0)*divu

863traceless_rateof_strain(Field3 v) {
864 Matrix S

866 S[0][0] = (2.0 / 3.0) * derx(v.x) - (1.0 / 3.0) * (dery(v.y) + derz(v.z))
