9u_dot_grad(Field3 f, Matrix m,real3 u){
10 suppress_unused_warning(f)
11 real3 res = m*u
12 if(AC_coordinate_system == AC_SPHERICAL_COORDINATES)
13 {
14 res.x = res.x - AC_INV_R*(u.y*f.y+u.z*f.z)
15 res.y = res.y + AC_INV_R*(u.y*f.x-u.z*f.z*AC_COT)
16 res.z = res.z + AC_INV_R*(u.z*f.x+u.z*f.y*AC_COT)
17 }
18 else if(AC_coordinate_system == AC_CYLINDRICAL_COORDINATES)
19 {
20 res.x = res.x - AC_INV_CYL_R*(u.y*f.y)
21 res.y = res.y + AC_INV_CYL_R*(u.y*f.x)
22 }
23 return res;
24}

402bij(Field3 v)
403{
404 d2A = get_d2A(v)
405
406 Matrix res;
407 res[0][0] = d2A[1][0][2] - d2A[2][0][1]
408 res[0][1] = d2A[1][1][2] - d2A[2][1][1]
409 res[0][2] = d2A[1][2][2] - d2A[2][2][1]
410
411 res[1][0] = d2A[2][0][0] - d2A[0][0][2]
412 res[1][1] = d2A[2][1][0] - d2A[0][1][2]
413 res[1][2] = d2A[2][2][0] - d2A[0][2][2]
414
415 res[2][0] = d2A[0][0][1] - d2A[1][0][0]
416 res[2][1] = d2A[0][1][1] - d2A[1][1][0]
417 res[2][2] = d2A[0][2][1] - d2A[1][2][0]
418
419 if(AC_coordinate_system == AC_SPHERICAL_COORDINATES)
420 {
421 res[3-1][2-1]=res[3-1][2-1]+dery(v.y)*AC_INV_R
422 res[2-1][3-1]=res[2-1][3-1]-derz(v.z)*AC_INV_R
423 res[1-1][3-1]=res[1-1][3-1]+derz(v.z)*AC_INV_R*AC_COT
424 res[3-1][1-1]=res[3-1][1-1]+derx(v.y)*AC_INV_R-v.y*(AC_INV_R*AC_INV_R)
425 res[2-1][1-1]=res[2-1][1-1]-derx(v.z)*AC_INV_R+v.z*(AC_INV_R*AC_INV_R)
426 res[1-1][2-1]=res[1-1][2-1]+dery(v.z)*AC_INV_R*AC_COT-v.z*(AC_INV_R*AC_INV_R)*(AC_INV_SIN_THETA*AC_INV_SIN_THETA)
427 }
428 else if(AC_coordinate_system == AC_CYLINDRICAL_COORDINATES)
429 {
430 res[3-1][2-1]=res[3-1][2-1]+ dery(v.y)*AC_INV_CYL_R
431 res[3-1][1-1]=res[3-1][1-1]+ derx(v.y)*AC_INV_CYL_R-v.y*(AC_INV_CYL_R*AC_INV_CYL_R)
432 }
433

774 d2A = get_d2A(s)
775 del2f = real3(
776 d2A[0][0][0] + d2A[1][1][0] + d2A[2][2][0],
777 d2A[0][0][1] + d2A[1][1][1] + d2A[2][2][1],
778 d2A[0][0][2] + d2A[1][1][2] + d2A[2][2][2]
779 )
780 if(AC_coordinate_system == AC_CYLINDRICAL_COORDINATES)
781 {
782 del2f.x -= (2*dery(s.y)+s.x)*AC_INV_CYL_R*AC_INV_CYL_R
783 del2f.y += (2*dery(s.x)-s.y)*AC_INV_CYL_R*AC_INV_CYL_R
784 }
785 if(AC_coordinate_system == AC_SPHERICAL_COORDINATES)
786 {
787 del2f.x += AC_INV_R*(
788 2*(
789 derx(s.x)-dery(s.y)-derz(s.z)
790 -AC_INV_R*(s.x+AC_COT*s.y)
791 )
792 +AC_COT*dery(s.x)
793 )
794 del2f.y += AC_INV_R*(
795 2*(
796 derx(s.y)-AC_COT*derz(s.z)+dery(s.x)
797 )
