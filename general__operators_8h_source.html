7u_dot_grad(Field3 f, Matrix m,real3 u){
8 suppress_unused_warning(f)
9 real3 res = m*u
10 if(AC_coordinate_system == AC_SPHERICAL_COORDINATES)
11 {
12 res.x = res.x - AC_INV_R*(u.y*f.y+u.z*f.z)
13 res.y = res.y + AC_INV_R*(u.y*f.x-u.z*f.z*AC_COT)
14 res.z = res.z + AC_INV_R*(u.z*f.x+u.z*f.y*AC_COT)
15 }
16 else if(AC_coordinate_system == AC_CYLINDRICAL_COORDINATES)
17 {
18 res.x = res.x - AC_INV_CYL_R*(u.y*f.y)
19 res.y = res.y + AC_INV_CYL_R*(u.y*f.x)
20 }
21 return res;
22}

381bij(Field3 v)
382{
383 d2A = get_d2A(v)
384
385 Matrix res;
386 res[0][0] = d2A[1][0][2] - d2A[2][0][1]
387 res[0][1] = d2A[1][1][2] - d2A[2][1][1]
388 res[0][2] = d2A[1][2][2] - d2A[2][2][1]
389
390 res[1][0] = d2A[2][0][0] - d2A[0][0][2]
391 res[1][1] = d2A[2][1][0] - d2A[0][1][2]
392 res[1][2] = d2A[2][2][0] - d2A[0][2][2]
393
394 res[2][0] = d2A[0][0][1] - d2A[1][0][0]
395 res[2][1] = d2A[0][1][1] - d2A[1][1][0]
396 res[2][2] = d2A[0][2][1] - d2A[1][2][0]
397
398 if(AC_coordinate_system == AC_SPHERICAL_COORDINATES)
399 {
400 res[3-1][2-1]=res[3-1][2-1]+dery(v.y)*AC_INV_R
401 res[2-1][3-1]=res[2-1][3-1]-derz(v.z)*AC_INV_R
402 res[1-1][3-1]=res[1-1][3-1]+derz(v.z)*AC_INV_R*AC_COT
403 res[3-1][1-1]=res[3-1][1-1]+derx(v.y)*AC_INV_R-v.y*(AC_INV_R*AC_INV_R)
404 res[2-1][1-1]=res[2-1][1-1]-derx(v.z)*AC_INV_R+v.z*(AC_INV_R*AC_INV_R)
405 res[1-1][2-1]=res[1-1][2-1]+dery(v.z)*AC_INV_R*AC_COT-v.z*(AC_INV_R*AC_INV_R)*(AC_INV_SIN_THETA*AC_INV_SIN_THETA)
406 }
407 else if(AC_coordinate_system == AC_CYLINDRICAL_COORDINATES)
408 {
409 res[3-1][2-1]=res[3-1][2-1]+ dery(v.y)*AC_INV_CYL_R
410 res[3-1][1-1]=res[3-1][1-1]+ derx(v.y)*AC_INV_CYL_R-v.y*(AC_INV_CYL_R*AC_INV_CYL_R)
411 }
412
413
414 return res;
415}

742laplace(Field3 s) {
743 d2A = get_d2A(s)
744 del2f = real3(
745 d2A[0][0][0] + d2A[1][1][0] + d2A[2][2][0],
746 d2A[0][0][1] + d2A[1][1][1] + d2A[2][2][1],
747 d2A[0][0][2] + d2A[1][1][2] + d2A[2][2][2]
748 )
749 if(AC_coordinate_system == AC_CYLINDRICAL_COORDINATES)
750 {
751 del2f.x -= (2*dery(s.y)+s.x)*AC_INV_CYL_R*AC_INV_CYL_R
752 del2f.y += (2*dery(s.x)-s.y)*AC_INV_CYL_R*AC_INV_CYL_R
753 }
754 if(AC_coordinate_system == AC_SPHERICAL_COORDINATES)
755 {
756 del2f.x += AC_INV_R*(
757 2*(
758 derx(s.x)-dery(s.y)-derz(s.z)
759 -AC_INV_R*(s.x+AC_COT*s.y)
760 )
761 +AC_COT*dery(s.x)
762 )
763 del2f.y += AC_INV_R*(
764 2*(
765 derx(s.y)-AC_COT*derz(s.z)+dery(s.x)
766 )
767 +AC_COT*dery(s.y)-AC_INV_R*(AC_INV_SIN_THETA*AC_INV_SIN_THETA)*s.y
768 )
769 del2f.z += AC_INV_R*(
770 2*(
771 derx(s.z)+derz(s.x)+AC_COT*derz(s.y)
772 )
773 +AC_COT*dery(s.z)-AC_INV_R*(AC_INV_SIN_THETA*AC_INV_SIN_THETA)*s.z
774 )
775 }
776 return del2f
777}

785traceless_strain(Matrix uij,divu)
786{
787 if(AC_coordinate_system == AC_SPHERICAL_COORDINATES || AC_coordinate_system == AC_CYLINDRICAL_COORDINATES)

789 fatal_error_message(true,"traceless_strain needs the velocity for curvilinear coordinates");

792 for row in 0:3{
793 sij[row][row] = uij[row][row] - (1.0/3.0)*divu

797 }
798 }
