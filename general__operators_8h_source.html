265bij(Field3 v)
266{
267 d2A = get_d2A(v)
268
269 Matrix res;
270 res[0][0] = d2A[1][0][2] - d2A[2][0][1]
271 res[0][1] = d2A[1][1][2] - d2A[2][1][1]
272 res[0][2] = d2A[1][2][2] - d2A[2][2][1]
273
274 res[1][0] = d2A[2][0][0] - d2A[0][0][2]
275 res[1][1] = d2A[2][1][0] - d2A[0][1][2]
276 res[1][2] = d2A[2][2][0] - d2A[0][2][2]
277
278 res[2][0] = d2A[0][0][1] - d2A[1][0][0]
279 res[2][1] = d2A[0][1][1] - d2A[1][1][0]
280 res[2][2] = d2A[0][2][1] - d2A[1][2][0]
281
282 if(AC_coordinate_system == AC_SPHERICAL_COORDINATES)
283 {
284 res[3-1][2-1]=res[3-1][2-1]+dery(v.y)*AC_INV_R
285 res[2-1][3-1]=res[2-1][3-1]-derz(v.z)*AC_INV_R
286 res[1-1][3-1]=res[1-1][3-1]+derz(v.z)*AC_INV_R*AC_COT
287 res[3-1][1-1]=res[3-1][1-1]+derx(v.y)*AC_INV_R-v.y*(AC_INV_R*AC_INV_R)
288 res[2-1][1-1]=res[2-1][1-1]-derx(v.z)*AC_INV_R+v.z*(AC_INV_R*AC_INV_R)
289 res[1-1][2-1]=res[1-1][2-1]+dery(v.z)*AC_INV_R*AC_COT-v.z*(AC_INV_R*AC_INV_R)*(AC_INV_SIN_THETA*AC_INV_SIN_THETA)
290 }
291 else if(AC_coordinate_system == AC_CYLINDRICAL_COORDINATES)
292 {
293 res[3-1][2-1]=res[3-1][2-1]+ dery(v.y)*AC_INV_CYL_R
294 res[3-1][1-1]=res[3-1][1-1]+ derx(v.y)*AC_INV_CYL_R-v.y*(AC_INV_CYL_R*AC_INV_CYL_R)
295 }
296
297
298 return res;
299}

524 d2A = get_d2A(s)
525 del2f = real3(
526 d2A[0][0][0] + d2A[1][1][0] + d2A[2][2][0],
527 d2A[0][0][1] + d2A[1][1][1] + d2A[2][2][1],
528 d2A[0][0][2] + d2A[1][1][2] + d2A[2][2][2]
529 )
530 if(AC_coordinate_system == AC_CYLINDRICAL_COORDINATES)
531 {
532 del2f.x -= (2*dery(s.y)+s.x)*AC_INV_CYL_R*AC_INV_CYL_R
533 del2f.y += (2*dery(s.x)-s.y)*AC_INV_CYL_R*AC_INV_CYL_R
534 }
535 if(AC_coordinate_system == AC_SPHERICAL_COORDINATES)
536 {
537 del2f.x += AC_INV_R*(
538 2*(
539 derx(s.x)-dery(s.y)-derz(s.z)
540 -AC_INV_R*(s.x+AC_COT*s.y)
541 )
542 +AC_COT*dery(s.x)
543 )
544 del2f.y += AC_INV_R*(
545 2*(
546 derx(s.y)-AC_COT*derz(s.z)+dery(s.x)
547 )
548 +AC_COT*dery(s.y)-AC_INV_R*(AC_INV_SIN_THETA*AC_INV_SIN_THETA)*s.y
549 )
550 del2f.z += AC_INV_R*(
551 2*(
552 derx(s.z)+derz(s.x)+AC_COT*derz(s.y)
553 )
554 +AC_COT*dery(s.z)-AC_INV_R*(AC_INV_SIN_THETA*AC_INV_SIN_THETA)*s.z
555 )
556 }
557 return del2f
